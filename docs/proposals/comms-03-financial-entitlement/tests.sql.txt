DO $$
DECLARE g uuid := (SELECT v::uuid FROM t_ctx WHERE k='growth'); a uuid := (SELECT v::uuid FROM t_ctx WHERE k='idA');

  EXECUTE 'SET LOCAL ROLE service_role';

  BEGIN
    r1 := public.comms_sms_claim(a, g, c1, 'bulksmsnigeria');
    r2 := public.comms_sms_claim(a, g, c2, 'bulksmsnigeria');
    r3 := public.comms_sms_settle(a, g, c2, 'sent', 'bulksmsnigeria', 'prov-A', NULL);
    r4 := public.comms_sms_settle(a, g, c1, 'sent', 'bulksmsnigeria', 'prov-A', NULL);
    r5 := public.comms_sms_settle(a, g, c1, 'failed', NULL, NULL, 'late');
    r6 := public.comms_sms_report_delivery(a, g, 'delivered', 'DELIVRD');
    r7 := public.comms_sms_report_delivery(a, g, 'failed', 'LATE');

  INSERT INTO t_result(name, ok, detail) VALUES ('F1b settle is compare-and-set on the claim held', coalesce(NOT r3 AND r4 AND NOT r5, false), d);
  INSERT INTO t_result(name, ok, detail) VALUES ('F1c delivery report moves sent to delivered once', coalesce(r6 AND NOT r7, false), d);
  INSERT INTO t_result(name, ok, detail) VALUES ('F1d service_role still cannot UPDATE the table directly', denied, 'direct UPDATE denied');

-- ===== B2 [01] refund policy =====
DO $$
DECLARE g uuid := (SELECT v::uuid FROM t_ctx WHERE k='growth'); a uuid := (SELECT v::uuid FROM t_ctx WHERE k='idA');
  b uuid := (SELECT v::uuid FROM t_ctx WHERE k='idB'); c uuid := (SELECT v::uuid FROM t_ctx WHERE k='idC'); dd uuid := (SELECT v::uuid FROM t_ctx WHERE k='idD');
  cb text := 'claim:' || gen_random_uuid()::text; cc text := 'claim:' || gen_random_uuid()::text; bal0 numeric := (SELECT v::numeric FROM t_ctx WHERE k='g_bal0');
  bal_b numeric; bal_c numeric; bal_d numeric; bal_op numeric; rb boolean; rc boolean; rd boolean; again jsonb; op jsonb; refused boolean := false; d text := '';
BEGIN
  EXECUTE 'SET LOCAL ROLE service_role';
  PERFORM set_config('request.jwt.claims','{"role":"service_role"}',true);
  BEGIN
    PERFORM public.comms_sms_claim(b, g, cb, 'bulksmsnigeria');
    rb := public.comms_sms_settle(b, g, cb, 'failed', NULL, NULL, 'provider_rejected', 'provider_rejected');
    bal_b := (SELECT balance FROM public.school_wallets WHERE school_id = g);
    again := public.comms_refund_unsubmitted(b, g, 'provider_rejected');
    PERFORM public.comms_sms_claim(c, g, cc, 'bulksmsnigeria');
    rc := public.comms_sms_settle(c, g, cc, 'failed', NULL, NULL, 'provider_error');
    bal_c := (SELECT balance FROM public.school_wallets WHERE school_id = g);
    rd := public.comms_sms_settle(dd, g, NULL, 'failed', NULL, NULL, 'blocked by consent', 'rejected_before_submission');
    bal_d := (SELECT balance FROM public.school_wallets WHERE school_id = g);
    op := public.comms_refund_unsubmitted(c, g, 'provider_rejected');
    bal_op := (SELECT balance FROM public.school_wallets WHERE school_id = g);
    BEGIN PERFORM public.comms_refund_unsubmitted(a, g, 'provider_rejected'); EXCEPTION WHEN others THEN refused := true; END;
  EXCEPTION WHEN undefined_function THEN d := 'functions missing';
  WHEN others THEN d := SQLERRM; END;
  EXECUTE 'RESET ROLE';
  INSERT INTO t_result(name, ok, detail) VALUES ('B2a provider-rejected message is refunded atomically with the close', coalesce(rb AND bal_b = bal0 - 40 + 10, false), d || ' balance ' || coalesce(bal_b::text,'?'));
  INSERT INTO t_result(name, ok, detail) VALUES ('B2b a second refund of the same message is a no-op', coalesce((again->>'refunded')::boolean = false AND again->>'reason' = 'already_refunded', false), d);
  INSERT INTO t_result(name, ok, detail) VALUES ('B2c an ambiguous failure (provider_error) is NOT auto-refunded', coalesce(rc AND bal_c = bal_b, false), d || ' balance ' || coalesce(bal_c::text,'?'));
  INSERT INTO t_result(name, ok, detail) VALUES ('B2d a rejection before submission is refunded', coalesce(rd AND bal_d = bal_c + 10, false), d || ' balance ' || coalesce(bal_d::text,'?'));
  INSERT INTO t_result(name, ok, detail) VALUES ('B2e an operator can refund a reconciled ambiguous failure', coalesce((op->>'refunded')::boolean AND bal_op = bal_d + 10, false), d || ' balance ' || coalesce(bal_op::text,'?'));
  INSERT INTO t_result(name, ok, detail) VALUES ('B2f a provider-accepted message can never be refunded', refused, 'refund of accepted message refused');
  INSERT INTO t_result(name, ok, detail) VALUES ('B2g net charge is exactly the one accepted message',
     coalesce((SELECT balance FROM public.school_wallets WHERE school_id = g) = bal0 - 10, false), 'wallet = start - 10');
END $$;
